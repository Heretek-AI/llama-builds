# P0 Sprint: CUDA Matrix + Binary CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Expand CUDA matrix from 2 to 5 builds and add a `llamaup` CLI for auto-detecting GPU and installing the right binary.

**Architecture:** Four new target directories with `extra_cmake_flags` for SM-specific builds, plus a universal build by removing explicit `CMAKE_CUDA_ARCHITECTURES`. The `llamaup` script fetches the manifest, detects GPU via `nvidia-smi`, maps compute capability to target, and installs.

**Tech Stack:** Bash (build scripts, CLI), Python (manifest generation, tests), JSON Schema (validation), GitHub Actions (CI matrix)

```bash
#!/usr/bin/env bash
# METADATA
# name=llama.cpp upstream CUDA (universal)
# repo=ggml-org/llama.cpp
# ref=0ab9d6fed73dbc5dc8026c868cb10a6728c4ed48
# backend=cuda
# arch=x86_64
# capabilities=chat,embed,flash-attn
set -euo pipefail

# Build llama.cpp CUDA (universal) from upstream.
# Uses llama.cpp's default CMAKE_CUDA_ARCHITECTURES which builds for
# all supported architectures via PTX fallback.

REPO="${REPO:-ggml-org/llama.cpp}"
REF="${REF:-main}"

echo "Building llama.cpp CUDA (universal)"
echo "  Repo: $REPO"
echo "  Ref:  $REF"
echo "  Backend: cuda"
echo "  Arch: x86_64"
echo "  CUDA Architectures: all (llama.cpp defaults)"

# For local builds (not in CI), clone and build manually
if [[ -z "${GITHUB_ACTIONS:-}" ]]; then
  echo "Running outside GitHub Actions — building locally..."

  BUILD_DIR=$(mktemp -d)
  trap 'rm -rf "$BUILD_DIR"' EXIT

  git clone --depth 1 --branch "$REF" "https://github.com/$REPO.git" "$BUILD_DIR/repo" 2>/dev/null \
    || git clone --depth 1 "https://github.com/$REPO.git" "$BUILD_DIR/repo"

  # Install CUDA toolkit if not present
  if ! command -v nvcc &> /dev/null; then
    echo "CUDA toolkit not found. Installing..."
    if [[ -f /etc/debian_version ]]; then
      sudo apt-get update && sudo apt-get install -y nvidia-cuda-toolkit ninja-build
    elif [[ -f /etc/redhat-release ]]; then
      sudo dnf install -y cuda-toolkit ninja-build
    else
      echo "ERROR: Cannot auto-install CUDA toolkit on this OS."
      echo "Please install CUDA toolkit and Ninja manually."
      exit 1
    fi
  else
    echo "CUDA toolkit found: $(nvcc --version | head -1)"
    if ! command -v ninja &> /dev/null; then
      echo "Installing Ninja build system..."
      if [[ -f /etc/debian_version ]]; then
        sudo apt-get install -y ninja-build
      elif [[ -f /etc/redhat-release ]]; then
        sudo dnf install -y ninja-build
      fi
    fi
  fi

  cd "$BUILD_DIR/repo"
  mkdir -p build && cd build
  # No explicit CMAKE_CUDA_ARCHITECTURES — use llama.cpp's universal defaults
  cmake .. -DGGML_CUDA=ON -DCMAKE_BUILD_TYPE=Release -G Ninja
  cmake --build . -j"$(nproc)"

  echo "Build complete. Binaries in: $(pwd)"
  ls -la llama-server llama-cli 2>/dev/null || echo "Note: binary names may vary"
else
  echo "Running in GitHub Actions — use the build-llama composite action."
  echo "See: https://github.com/Heretek-AI/llama-builds/actions"
fi
```

```bash
#!/usr/bin/env bash
# llamaup - Binary distribution CLI for llama.cpp builds
# Usage: curl -sSL https://heretek-ai.github.io/llama-builds/llamaup | bash
#
# Auto-detects GPU and installs the right binary from our manifest.

set -euo pipefail

# Configuration
MANIFEST_URL="${LLAMAUP_MANIFEST_URL:-https://heretek-ai.github.io/llama-builds/manifest.json}"
INSTALL_DIR="${LLAMAUP_INSTALL_DIR:-$HOME/.local/bin}"
GPU_MAP_URL="${LLAMAUP_GPU_MAP_URL:-https://raw.githubusercontent.com/Heretek-AI/llama-builds/main/configs/gpu_map.json}"

# Colors
RED='\033[0;31m'
GREEN='\033[0;32m'
YELLOW='\033[1;33m'
NC='\033[0m' # No Color

usage() {
    cat <<EOF
llamaup - Install pre-built llama.cpp binaries

Usage: llamaup [OPTIONS]

Options:
  --version VERSION    Install a specific version (e.g., abc1234-1)
  --list               List available builds
  --dry-run            Show what would be installed without installing
  --install-dir DIR    Install to DIR (default: ~/.local/bin)
  --help               Show this help message

Examples:
  llamaup                    # Auto-detect GPU and install
  llamaup --list             # See available builds
  llamaup --dry-run          # Preview installation
  llamaup --version abc1234-1  # Install specific version

Environment Variables:
  LLAMAUP_MANIFEST_URL      Override manifest URL
  LLAMAUP_INSTALL_DIR       Override install directory
  LLAMAUP_GPU_MAP_URL       Override GPU map URL
EOF
}

log_info() {
    echo -e "${GREEN}[INFO]${NC} $1"
}

log_warn() {
    echo -e "${YELLOW}[WARN]${NC} $1"
}

log_error() {
    echo -e "${RED}[ERROR]${NC} $1"
}

# Check for required commands
check_deps() {
    local missing=()
    for cmd in curl jq; do
        if ! command -v "$cmd" &> /dev/null; then
            missing+=("$cmd")
        fi
    done
    if [[ ${#missing[@]} -gt 0 ]]; then
        log_error "Missing required commands: ${missing[*]}"
        log_error "Install them with: sudo apt-get install ${missing[*]}"
        exit 1
    fi
}

# Fetch JSON from URL
fetch_json() {
    local url="$1"
    curl -sSL --fail "$url" 2>/dev/null
}

# Detect NVIDIA GPU compute capability
detect_nvidia_gpu() {
    if ! command -v nvidia-smi &> /dev/null; then
        return 1
    fi

    local compute_cap
    compute_cap=$(nvidia-smi --query-gpu=compute_cap --format=csv,noheader 2>/dev/null | head -1)

    if [[ -z "$compute_cap" ]]; then
        return 1
    fi

    # Convert "8.9" format to "8.9" (already correct)
    echo "$compute_cap"
}

# Detect AMD GPU
detect_amd_gpu() {
    if ! command -v rocm-smi &> /dev/null; then
        return 1
    fi

    # ROCm detection - return "rocm" as indicator
    echo "rocm"
}

# Map compute capability to target slug
map_gpu_to_target() {
    local compute_cap="$1"
    local gpu_map_json="$2"

    # Try exact match first
    local target
    target=$(echo "$gpu_map_json" | jq -r --arg cap "$compute_cap" '.[$cap] // empty')

    if [[ -n "$target" ]]; then
        echo "$target"
        return 0
    fi

    # Try major version match (e.g., "8.9" -> "8")
    local major
    major=$(echo "$compute_cap" | cut -d. -f1)
    target=$(echo "$gpu_map_json" | jq -r --arg cap "$major" '.[$cap] // empty')

    if [[ -n "$target" ]]; then
        echo "$target"
        return 0
    fi

    # Fall back to default
    echo "$gpu_map_json" | jq -r '.default'
}

# Get artifact info from manifest
get_artifact_info() {
    local manifest_json="$1"
    local target_slug="$2"
    local version="${3:-}"

    if [[ -n "$version" ]]; then
        # Specific version requested
        echo "$manifest_json" | jq -r --arg slug "$target_slug" --arg ver "$version" \
            '.targets[$slug] | select(.version == $ver) | {artifact: .build.artifact, version: .version}'
    else
        # Latest version
        echo "$manifest_json" | jq -r --arg slug "$target_slug" \
            '.targets[$slug] | {artifact: .build.artifact, version: .version}'
    fi
}

# Download and verify artifact
download_artifact() {
    local artifact_name="$1"
    local version="$2"

    # Construct download URL (GitHub Releases format)
    local download_url="https://github.com/Heretek-AI/llama-builds/releases/download/v${version}/${artifact_name}"

    log_info "Downloading: $artifact_name"

    local tmp_dir
    tmp_dir=$(mktemp -d)
    trap 'rm -rf "$tmp_dir"' EXIT

    if ! curl -sSL --fail -o "$tmp_dir/$artifact_name" "$download_url"; then
        log_error "Failed to download artifact"
        return 1
    fi

    echo "$tmp_dir/$artifact_name"
}

# Install binary
install_binary() {
    local artifact_path="$1"

    mkdir -p "$INSTALL_DIR"

    log_info "Installing to: $INSTALL_DIR"

    # Extract archive
    tar -xzf "$artifact_path" -C "$INSTALL_DIR"

    # Make binaries executable
    chmod +x "$INSTALL_DIR"/llama-* 2>/dev/null || true

    log_info "Installation complete!"
    log_info "Binaries installed to: $INSTALL_DIR"
    log_info "Add to PATH: export PATH=\"$INSTALL_DIR:\$PATH\""
}

# List available builds
list_builds() {
    local manifest_json="$1"

    echo "Available builds:"
    echo ""
    echo "$manifest_json" | jq -r '.targets | to_entries[] | select(.value.backend == "cuda") | "\(.key)\t\(.value.name)\t\(..value.version)\t\(.value.build.artifact)"' | \
        column -t -s$'\t'
}

# Main function
main() {
    local version=""
    local list_only=false
    local dry_run=false

    # Parse arguments
    while [[ $# -gt 0 ]]; do
        case "$1" in
            --version)
                version="$2"
                shift 2
                ;;
            --list)
                list_only=true
                shift
                ;;
            --dry-run)
                dry_run=true
                shift
                ;;
            --install-dir)
                INSTALL_DIR="$2"
                shift 2
                ;;
            --help|-h)
                usage
                exit 0
                ;;
            *)
                log_error "Unknown option: $1"
                usage
                exit 1
                ;;
        esac
    done

    # Check dependencies
    check_deps

    # Fetch manifest
    log_info "Fetching manifest..."
    local manifest_json
    manifest_json=$(fetch_json "$MANIFEST_URL") || {
        log_error "Failed to fetch manifest from $MANIFEST_URL"
        exit 1
    }

    # List mode
    if [[ "$list_only" == true ]]; then
        list_builds "$manifest_json"
        exit 0
    fi

    # Fetch GPU map
    log_info "Fetching GPU map..."
    local gpu_map_json
    gpu_map_json=$(fetch_json "$GPU_MAP_URL") || {
        log_error "Failed to fetch GPU map from $GPU_MAP_URL"
        exit 1
    }

    # Detect GPU
    local compute_cap
    local target_slug

    if compute_cap=$(detect_nvidia_gpu); then
        log_info "Detected NVIDIA GPU with compute capability: $compute_cap"
        target_slug=$(map_gpu_to_target "$compute_cap" "$gpu_map_json")
    elif compute_cap=$(detect_amd_gpu); then
        log_warn "AMD GPU detected — CUDA builds may not work"
        target_slug="upstream-cuda"  # Fall back to universal
    else
        log_warn "No GPU detected — using universal build"
        target_slug="upstream-cuda"
    fi

    log_info "Selected target: $target_slug"

    # Get artifact info
    local artifact_info
    artifact_info=$(get_artifact_info "$manifest_json" "$target_slug" "$version") || {
        log_error "Target '$target_slug' not found in manifest"
        exit 1
    }

    local artifact_name
    artifact_name=$(echo "$artifact_info" | jq -r '.artifact')
    local artifact_version
    artifact_version=$(echo "$artifact_info" | jq -r '.version')

    if [[ -z "$artifact_name" || "$artifact_name" == "null" ]]; then
        log_error "No artifact found for target '$target_slug'"
        exit 1
    fi

    log_info "Artifact: $artifact_name"
    log_info "Version: $artifact_version"

    # Dry run mode
    if [[ "$dry_run" == true ]]; then
        echo ""
        echo "Dry run — would install:"
        echo "  Target: $target_slug"
        echo "  Artifact: $artifact_name"
        echo "  Version: $artifact_version"
        echo "  Install to: $INSTALL_DIR"
        exit 0
    fi

    # Download and install
    local artifact_path
    artifact_path=$(download_artifact "$artifact_name" "$artifact_version") || {
        log_error "Failed to download artifact"
        exit 1
    }

    install_binary "$artifact_path"

    log_info "Done! Run 'llama-cli --version' to verify."
}

main "$@"
```

```markdown
## CUDA Builds

We provide pre-built CUDA binaries for common GPU architectures:

| Build | SM | GPUs | Notes |
|-------|-----|------|-------|
| `upstream-cuda` | universal | All CUDA GPUs | PTX fallback, works everywhere |
| `upstream-cuda-sm80` | sm_80 | A100, A30 | Optimized for datacenter Ampere |
| `upstream-cuda-sm86` | sm_86 | RTX 30xx, A10 | Optimized for consumer Ampere |
| `upstream-cuda-sm89` | sm_89 | RTX 40xx, L4, L40S | Optimized for Ada Lovelace |
| `upstream-cuda-sm90` | sm_90 | H100, H200 | Optimized for Hopper |

The universal build uses llama.cpp's default `CMAKE_CUDA_ARCHITECTURES`,
which compiles for all supported architectures via PTX fallback. This means
one binary works on any CUDA GPU, but may have JIT overhead on first run.

SM-specific builds are optimized for exact architectures, avoiding JIT
overhead and producing smaller binaries.
```
